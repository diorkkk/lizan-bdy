SUPABASE SETUP · ADMIN AUTH · PRIVATE WISH DATABASE 
-- Run in Supabase SQL Editor. Safe to re-run: existing policies are dropped and recreated.
create table if not exists public.admin_users (
  user_id uuid primary key references auth.users(id) on delete cascade
);
create table if not exists public.private_wishes (
  id bigint generated always as identity primary key,
  visitor_name text not null default 'Anonymous',
  wish_text text not null check (char_length(wish_text) between 1 and 1000),
  created_at timestamptz not null default now()
);
create table if not exists public.birthday_settings (
  id integer primary key check (id = 1),
  birthday_at timestamptz not null
);
insert into public.birthday_settings(id,birthday_at)
values (1,'2026-12-13T00:00:00+05:45')
on conflict (id) do nothing;
create table if not exists public.visitor_events (
  id bigint generated always as identity primary key,
  page text not null default 'birthday',
  created_at timestamptz not null default now()
);

alter table public.admin_users enable row level security;
alter table public.private_wishes enable row level security;
alter table public.birthday_settings enable row level security;
alter table public.visitor_events enable row level security;

create or replace function public.is_birthday_admin()
returns boolean language sql stable security definer set search_path=public
as $$ select exists(select 1 from public.admin_users where user_id=auth.uid()) $$;

drop policy if exists "admin can verify own role" on public.admin_users;
create policy "admin can verify own role" on public.admin_users for select to authenticated
using (user_id = auth.uid());
drop policy if exists "visitors can submit wishes" on public.private_wishes;
create policy "visitors can submit wishes" on public.private_wishes for insert to anon, authenticated
with check (char_length(wish_text) between 1 and 1000);
drop policy if exists "only admin reads wishes" on public.private_wishes;
create policy "only admin reads wishes" on public.private_wishes for select to authenticated
using (public.is_birthday_admin());
drop policy if exists "public reads birthday date" on public.birthday_settings;
create policy "public reads birthday date" on public.birthday_settings for select to anon, authenticated using (true);
drop policy if exists "admin inserts birthday date" on public.birthday_settings;
create policy "admin inserts birthday date" on public.birthday_settings for insert to authenticated
with check (public.is_birthday_admin());
drop policy if exists "admin updates birthday date" on public.birthday_settings;
create policy "admin updates birthday date" on public.birthday_settings for update to authenticated
using (public.is_birthday_admin()) with check (public.is_birthday_admin());
drop policy if exists "visitors record visit" on public.visitor_events;
create policy "visitors record visit" on public.visitor_events for insert to anon, authenticated
with check (page = 'birthday');
drop policy if exists "admin reads visitor stats" on public.visitor_events;
create policy "admin reads visitor stats" on public.visitor_events for select to authenticated
using (public.is_birthday_admin());
